   10  async function requireUser(ctx: {
   11    auth: {
   12      getUserIdentity: () => Promise<{ tokenIdentifier: string } | null>;
   13    };
   14    db: import("./_generated/server").DatabaseReader;
   15  }) {
   16    const identity = await ctx.auth.getUserIdentity();
   17  
   18    if (!identity) {
   19      throw new ConvexError({
   20        message: "Non authentifié",
   21        code: "UNAUTHENTICATED",
   22      });
   23    }
   24  
   25    const user = await ctx.db
   26      .query("users")
   27      .withIndex("by_token", (q) =>
   28        q.eq("tokenIdentifier", identity.tokenIdentifier),
   29      )
   30      .unique();
   31  
   32    if (!user) {
   33      throw new ConvexError({
   34        message: "Utilisateur introuvable",
   35        code: "NOT_FOUND",
   36      });
   37    }
   38  
   39    return user;
   40  }

   44  /** List all conversations for the current user, ordered by most recent */
   45  export const listConversations = query({
   46    args: {},
   47  
   48    handler: async (
   49      ctx,
   50    ): Promise<
   51      Array<{
   52        conversationId: Id<"conversations">;
   53        isGroup: boolean;
   54        groupName?: string;
   55        groupAvatar?: string;
   56        lastMessageText?: string;
   57        lastMessageSenderId?: Id<"users">;
   58        updatedAt: string;
   59        unreadCount: number;
   60        otherParticipant?: {
   61          _id: Id<"users">;
   62          name?: string;
   63          avatar?: string;
   64        };
   65        participantIds: Id<"users">[];
   66      }>
   67    > => {
   68      const user = await requireUser(ctx);
   69  
   70      const memberships = await ctx.db
   71        .query("conversationMembers")
   72        .withIndex("by_user", (q) => q.eq("userId", user._id))
   73        .collect();
   74  
   75      const results = await Promise.all(
   76        memberships.map(async (m) => {
   77          const conv = await ctx.db.get(m.conversationId);
   78  
   79          if (!conv) return null;
   80  
   81          let otherParticipant:
   82            | {
   83                _id: Id<"users">;
   84                name?: string;
   85                avatar?: string;
   86              }
   87            | undefined;
   88  
   89          if (!conv.isGroup) {
   90            const otherId = conv.participantIds.find((id) => id !== user._id);
   91  
   92            if (otherId) {
   93              const other = await ctx.db.get(otherId);
   94  
   95              if (other) {
   96                otherParticipant = {
   97                  _id: other._id,
   98                  name: other.name,
   99                  avatar: other.avatar,
  100                };
  101              }
  102            }
  103          }
  104  
  105          return {
  106            conversationId: conv._id,
  107            isGroup: conv.isGroup,
  108            groupName: conv.groupName,
  109            groupAvatar: conv.groupAvatar,
  110            lastMessageText: conv.lastMessageText,
  111            lastMessageSenderId: conv.lastMessageSenderId,
  112            updatedAt: conv.updatedAt,
  113            unreadCount: m.unreadCount,
  114            otherParticipant,
  115            participantIds: conv.participantIds,
  116          };
  117        }),
  118      );
  119  
  120      return results
  121        .filter((r): r is NonNullable<typeof r> => r !== null)
  122        .sort((a, b) => (a.updatedAt < b.updatedAt ? 1 : -1));
  123    },
  124  });

  918  /** Set typing status for the current user in a conversation */
  919  export const setTyping = mutation({
  920    args: {
  921      conversationId: v.id("conversations"),
  922      isTyping: v.boolean(),
  923    },
  924  
  925    handler: async (ctx, args): Promise<void> => {
  926      const user = await requireUser(ctx);
  927  
  928      const existing = await ctx.db
  929        .query("typingStatus")
  930        .withIndex("by_user_and_conversation", (q) =>
  931          q.eq("userId", user._id).eq("conversationId", args.conversationId),
  932        )
  933        .unique();
  934  
  935      if (args.isTyping) {
  936        const now = new Date().toISOString();
  937  
  938        if (existing) {
  939          await ctx.db.patch(existing._id, {
  940            updatedAt: now,
  941          });
  942        } else {
  943          await ctx.db.insert("typingStatus", {
  944            conversationId: args.conversationId,
  945            userId: user._id,
  946            updatedAt: now,
  947          });
  948        }
  949      } else if (existing) {
  950        await ctx.db.delete(existing._id);
  951      }
  952    },
  953  });

  967        .query("typingStatus")
  968        .withIndex("by_conversation", (q) =>

  971        .collect();

  973      const active = rows.filter(
  974        (r) => r.userId !== user._id && r.updatedAt > cutoff,
